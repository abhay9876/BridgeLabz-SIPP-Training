boolean solveMaze(int[][] maze, int x, int y, int[][] solution) {
    if (x == N-1 && y == N-1) {
        solution[x][y] = 1;
        return true;
    }

    if (isSafe(maze, x, y)) {
        solution[x][y] = 1;

        if (solveMaze(maze, x+1, y, solution)) return true;
        if (solveMaze(maze, x, y+1, solution)) return true;

        solution[x][y] = 0;
        return false;
    }
    return false;
}
