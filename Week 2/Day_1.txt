2. Movie Ticket Booking

class Ticket {
    String movieName;
    String seatNumber;
    double price;
    boolean booked;
}

class Theatre {
    List<Ticket> tickets;

    Ticket issueTicket(String movieName, String seatNumber) {
        for (Ticket t : tickets) {
            if (t.seatNumber == seatNumber && !t.booked) {
                t.booked = true;
                print("Ticket issued for " + movieName + " Seat " + seatNumber);
                return t;
            }
        }
        print("Seat not available");
        return null;
    }

    void cancelTicket(Ticket ticket) {
        ticket.booked = false;
        print("Ticket for " + ticket.seatNumber + " cancelled.");
    }
}

class Customer {
    String name;

    Ticket bookTicket(Theatre theatre, String movie, String seat) {
        return theatre.issueTicket(movie, seat);
    }

    void cancelTicket(Theatre theatre, Ticket ticket) {
        theatre.cancelTicket(ticket);
    }
}
